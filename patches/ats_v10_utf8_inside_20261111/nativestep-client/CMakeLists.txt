cmake_minimum_required(VERSION 3.20)
project(ClientInput LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(step_core STATIC src/step_core.cpp)
target_include_directories(step_core PUBLIC include)
add_executable(step_core_test tests/step_core_test.cpp)
target_link_libraries(step_core_test PRIVATE step_core)
enable_testing()
add_test(NAME step_core COMMAND step_core_test)
if(WIN32)
  set(DINPUT_EXPORTS dinput.def)
  if(MINGW)
    set(DINPUT_EXPORTS dinput.mingw.def)
  endif()
  add_library(dinput SHARED src/ats_state.cpp src/ats_hud.cpp src/nativestep_dll.cpp src/mega_client.cpp src/client_bootstrap.cpp
      src/dinput_proxy.cpp src/step_network.cpp ${DINPUT_EXPORTS})
  target_include_directories(dinput PRIVATE include)
  target_link_libraries(dinput PRIVATE step_core gdiplus gdi32 user32 ws2_32 iphlpapi advapi32)
  target_compile_definitions(dinput PRIVATE UNICODE _UNICODE)
  add_executable(ATS_Preview WIN32 src/ats_state.cpp src/ats_hud.cpp tests/ats_preview.cpp)
  target_include_directories(ATS_Preview PRIVATE include)
  target_link_libraries(ATS_Preview PRIVATE gdiplus gdi32 user32)
  add_executable(win_host tests/win_host.cpp)
  target_link_libraries(win_host PRIVATE user32)
  add_test(NAME unsupported_host COMMAND win_host)
  set_tests_properties(unsupported_host PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:dinput>")
endif()

add_executable(ats_state_test src/ats_state.cpp tests/ats_state_test.cpp)
target_include_directories(ats_state_test PRIVATE include)
add_test(NAME ats_state COMMAND ats_state_test)

add_executable(input_scope_test tests/input_scope_test.cpp)
target_link_libraries(input_scope_test PRIVATE step_core)
add_test(NAME input_scope COMMAND input_scope_test)
if(WIN32)
  add_executable(background_adapter_test src/nativestep_dll.cpp tests/background_adapter_test.cpp)
  target_link_libraries(background_adapter_test PRIVATE step_core user32)
  add_test(NAME background_adapter COMMAND background_adapter_test)
endif()
