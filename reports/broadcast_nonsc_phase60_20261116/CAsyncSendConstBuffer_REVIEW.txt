43:      * The single shared-broadcast SEND contract.
44:      * Native <=16 KiB: retain one immutable ConstBuffer source per owner.
45:      * Other transports/large frames: enqueue those SAME frozen bytes directly.
46:      * No recipient-specific packet construction, no legacy serializer.
47:      * true means this socket actually admitted a SEND lease.
48:      */
49:     public final synchronized boolean send(CUserSocket target,int recipientId) {
50:         if(closed)throw new IllegalStateException("broadcast already closed");
51:         if(target==null || exclude(recipientId) || !target.isClientConnected())return false;
52:         final int limit=recipientId==senderId?Integer.MAX_VALUE:pendingLimit;
53:         if(wire.length<=16384 && target.supportsConstBuffer()
54:                 && target.pendingSendBytes()<limit) {
55:             if(source==null) {
56:                 try {
57:                     source=target.createConstBuffer(wire);
58:                 } catch (RuntimeException failure) {
59:                     // Resource acquisition failed before any SEND was staged.
60:                     // The same frozen bytes remain available to the FIFO path.
61:                 }
62:             }
63:             if(source!=null && target.ownsConstBuffer(source)
64:                     && target.enqueueConstBufferConfirmed(source,limit))
65:                 return true;
66:         }
67:         return target.sendFrozenWire(wire);
68:     }

70:     @Override public final synchronized void close() {
71:         if (closed) return;
72:         closed = true;
73:         if (source != null) { source.release(); source = null; }
74:     }

85:         public Source(CIOServer owner, MemorySegment handle, byte[] bytes) {
86:             this.owner = owner;
87:             this.handle = handle;
88:             wire = ByteBuffer.wrap(bytes).asReadOnlyBuffer().order(ByteOrder.LITTLE_ENDIAN);
89:         }
90:         public Source retain() {
91:             for (;;) {
92:                 int n = references.get();
93:                 if (n == 0 || n == Integer.MAX_VALUE) throw new IllegalStateException("constant source lifetime");
94:                 if (references.compareAndSet(n, n + 1)) return this;
95:             }
96:         }
97:         public void release() {
98:             int n = references.decrementAndGet();
99:             if (n < 0) throw new IllegalStateException("constant source double release");
100:             if (n == 0) owner.releaseConstBuffer(handle);
101:         }
