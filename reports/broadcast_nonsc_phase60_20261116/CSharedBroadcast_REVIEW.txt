16:     public static void send(L1Object anchor, byte[] wire, boolean fast, int radius,
17:             Predicate<L1PcInstance> filter) {
18:         if (anchor == null || wire == null) return;
19:         final L1World world=L1World.getInstance();
20:         final boolean borrowed=fast && radius < 0;
21:         final List<L1PcInstance> recipients=borrowed
22:                 ? world.borrowVisiblePlayerFast(anchor)
23:                 : radius < 0 ? world.getVisiblePlayer(anchor)
24:                 : world.getVisiblePlayer(anchor, radius);
25:         try (var shared=new CAsyncSendConstBufferExceptSelf(wire,anchor.getId(),Integer.MAX_VALUE)) {
26:             for (L1PcInstance pc:recipients) {
27:                 if (!CTLMFilter.sameWorldOther(anchor,pc)
28:                         || filter != null && !filter.test(pc))continue;
29:                 final CIOClient client=pc.getNetConnection();
30:                 if(client!=null && client.isConnected())
31:                     shared.send(client.getConnection(),pc.getId());
32:             }
33:         } finally {
34:             if(borrowed)world.releaseVisiblePlayerFast(recipients);
35:         }
36:     }

75:     /** Proto.sendRaw uses an unframed already-encoded payload. Convert once
76:      * to its 2-byte little-endian wire length, then broadcast shared bytes.
77:      */
78:     private static byte[] frameRaw(byte[] encoded) {
79:         if(encoded==null || encoded.length==0 || encoded.length>65533)return null;
80:         final int size=encoded.length+2;
81:         final byte[] frame=new byte[size];
82:         frame[0]=(byte)size;
83:         frame[1]=(byte)(size>>>8);
84:         System.arraycopy(encoded,0,frame,2,encoded.length);
85:         return frame;
86:     }

176:     /**
177:      * Send identical frames in recipient-major order. Previous behavior was
178:      * for each visible PC: send all display effects in input array order.
179:      * Team filtering, map iteration order, and socket FIFO are unchanged.
180:      * This method never serializes inside the recipient iteration.
181:      */
182:     public static void sendMapTeamFrames(
183:             int mapId, int teamId, boolean teamOnly, byte[][] wires) {
184:         if (wires==null || wires.length==0) return;
185:         final CAsyncSendConstBuffer[] shared=
186:                 new CAsyncSendConstBuffer[wires.length];
187:         int opened=0;
188:         try {
189:             for(int i=0;i<wires.length;i++) {
190:                 if(wires[i]==null) return;
191:                 shared[i]=new CAsyncSendConstBuffer(
192:                         wires[i],Integer.MIN_VALUE,Integer.MAX_VALUE);
193:                 opened++;
194:             }
195:             for(L1PcInstance pc:L1World.getInstance().getVisiblePlayers(mapId)) {
196:                 if(pc==null || (teamOnly && pc.get_current_combat_team_id()!=teamId))
197:                     continue;
198:                 final CIOClient client=pc.getNetConnection();
199:                 if(client==null || !client.isConnected()) continue;
200:                 final var socket=client.getConnection();
201:                 if(socket==null || !socket.isClientConnected()) continue;
202:                 for(int i=0;i<opened;i++)
203:                     shared[i].send(socket,pc.getId());
204:             }
205:         } finally {
206:             for(int i=0;i<opened;i++) shared[i].close();
207:         }
208:     }
