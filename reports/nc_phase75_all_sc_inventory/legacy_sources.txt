package l1j.server.server.model.network.core.sc.packet;

import java.util.ArrayList;
import java.util.concurrent.atomic.AtomicInteger;
import l1j.server.server.ActionCodes;
import l1j.server.server.Opcodes;
import l1j.server.server.model.L1Character;
import l1j.server.server.model.L1World;
import l1j.server.server.model.Instance.L1PcInstance;
import l1j.server.server.model.network.buffer.WriteBuffer;
import l1j.server.server.model.network.core.connection.CUserSocket;
import l1j.server.server.model.network.core.connection.CIOFrame;
import l1j.server.server.model.network.core.session.CIOClient;

public final class SC_AreaAttackEffect {
    private static final AtomicInteger SEQUENCE = new AtomicInteger();
    private SC_AreaAttackEffect() {}

    public static void sendNoDirAndBroadcast(L1Character caster, ArrayList<L1Character> targets,
                                              int gfx, int action) {
        if(caster==null)return;
        final int id=caster.getId(),x=caster.getX(),y=caster.getY();
        final int heading=caster.getHeading();
        final int count=targets==null?0:targets.size();
        final int[] ids=new int[count];
        for(int i=0;i<count;i++) {
            final L1Character target=targets.get(i);
            if(target!=null)ids[i]=target.getId();
        }
        final int seq=SEQUENCE.incrementAndGet();
        final byte[] wire=sharedWire(id,x,y,heading,seq,gfx,action,ids);
        if(wire==null)return;
        if(wire.length>16384) {
            final byte[] payload=java.util.Arrays.copyOfRange(wire,2,wire.length);
            if(caster instanceof L1PcInstance)
                sendLarge((L1PcInstance)caster,wire.length,payload);
            for(L1PcInstance viewer:L1World.getInstance().getVisiblePlayer(caster))
                sendLarge(viewer,wire.length,payload);
            return;
        }
        try(var shared=new l1j.server.server.model.network.core.broadcast.CAsyncSendConstBuffer(
                wire,Integer.MIN_VALUE,Integer.MAX_VALUE)) {
            // Preserve the old self-first emission even if the AOI query also
            // contains that player; duplicate recipients stay duplicated.
            if(caster instanceof L1PcInstance)
                sendShared(shared,(L1PcInstance)caster);
            for(L1PcInstance viewer:L1World.getInstance().getVisiblePlayer(caster))
                sendShared(shared,viewer);
        }
    }

    /** Original SC_AreaAttackEffect::send WriteBuffer byte layout. One buffer
     * per attack; per-socket encryption and SEND FIFO remain native-owned.
     */
    public static byte[] sharedWire(int id,int x,int y,int heading,int seq,int gfx,
                                    int action,int[] ids) {
        if(ids==null)return null;
        final long length=26L+6L*ids.length;
        if(length>0xffff)return null;
        final int size=(int)length;
        final byte[] wire=new byte[size];
        put16(wire,0,size);
        wire[2]=(byte)Opcodes.S_ATTACK_MANY;
        wire[3]=(byte)action;
        put32(wire,4,id);
        put16(wire,8,x);
        put16(wire,10,y);
        wire[12]=(byte)heading;
        put32(wire,13,seq);
        put16(wire,17,gfx);
        // 19: byte 0; 20-21: short 0
        put16(wire,22,ids.length);
        int at=24;
        for(int targetId:ids) {
            put32(wire,at,targetId);at+=4;
            wire[at++]=0x20;
            wire[at++]=(byte)ActionCodes.ACTION_Damage;
        }
        // at..at+1: terminal short 0
        return wire;
    }

    private static void put16(byte[] wire,int at,int value) {
        wire[at]=(byte)value;
        wire[at+1]=(byte)(value>>>8);
    }
    private static void put32(byte[] wire,int at,int value) {
        wire[at]=(byte)value;
        wire[at+1]=(byte)(value>>>8);
        wire[at+2]=(byte)(value>>>16);
        wire[at+3]=(byte)(value>>>24);
    }
    /** Full 65,535-byte game frames use the existing large direct FIFO lease.
     * The encoded body is shared across all recipients; only the mandatory
     * per-socket buffer copy is performed, and the original encoder is gone.
     */
    private static boolean sendLarge(L1PcInstance pc,int length,byte[] payload) {
        if(pc==null)return false;
        final CIOClient client=pc.getNetConnection();
        if(client==null || !client.isConnected())return false;
        final CUserSocket connection=client.getConnection();
        if(connection==null || !connection.isClientConnected())return false;
        final CIOFrame frame=connection.beginDirectFrame(length);
        if(frame==null)return false;
        boolean committed=false;
        try {
            frame.buffer().writeBytes(payload);
            committed=connection.commitDirectFrame(frame);
            return committed;
        } finally {
            if(!committed)connection.abortDirectFrame(frame);
        }
    }

    private static void sendShared(
            l1j.server.server.model.network.core.broadcast.CAsyncSendConstBuffer shared,
            L1PcInstance pc) {
        if(pc==null)return;
        final CIOClient client=pc.getNetConnection();
        if(client==null || !client.isConnected())return;
        final CUserSocket socket=client.getConnection();
        if(socket!=null && socket.isClientConnected())
            shared.send(socket,pc.getId());
    }

    
}

package l1j.server.server.model.network.core.sc.packet;

import java.util.concurrent.atomic.AtomicInteger;
import l1j.server.server.Opcodes;
import l1j.server.server.model.Instance.L1PcInstance;
import l1j.server.server.model.L1World;
import l1j.server.server.model.network.buffer.WriteBuffer;
import l1j.server.server.model.network.core.connection.CUserSocket;
import l1j.server.server.model.network.core.connection.CIOFrame;
import l1j.server.server.model.network.core.session.CIOClient;

public final class SC_Attack {
    private static final AtomicInteger SEQ = new AtomicInteger();
    private SC_Attack() {}

    private static byte[] sharedWire(int attackerId, int targetId, int damage, int heading,
            int seq, int effect, int x, int y, int endX, int endY) {
        return java.nio.ByteBuffer.allocate(39).order(java.nio.ByteOrder.LITTLE_ENDIAN)
                .putShort((short) 39).put((byte) l1j.server.server.Opcodes.S_ATTACK)
                .put((byte) 1).putInt(attackerId).putInt(targetId)
                .putShort((short) damage).put((byte) heading).putInt(seq)
                .putShort((short) effect).put((byte) 0)
                .putShort((short) x).putShort((short) y)
                .putShort((short) endX).putShort((short) endY)
                .putInt(0).putShort((short) 0).put((byte) 0)
                .putShort((short) 0).array();
    }

    public static void sendAndBroadcastFast(L1PcInstance pc, int targetId, int effect,
                                            int damage, int heading, int endX, int endY) {
        if (pc == null) return;
        final int attackerId = pc.getId();
        final int x = pc.getX();
        final int y = pc.getY();
        final int seq = SEQ.incrementAndGet();
        send(pc, attackerId, targetId, damage, heading, seq, effect, x, y, endX, endY);
        l1j.server.server.model.network.core.broadcast.CSharedBroadcast.send(pc, sharedWire(attackerId, targetId, damage, heading, seq, effect, x, y, endX, endY), true, -1, null);
    }

    public static boolean send(L1PcInstance pc, int attackerId, int targetId,
                               int damage, int heading, int seq, int effect,
                               int x, int y, int endX, int endY) {
        return pc != null && send(pc.getNetConnection(), attackerId, targetId, damage, heading,
                                  seq, effect, x, y, endX, endY);
    }

    public static boolean send(CIOClient client, int attackerId, int targetId,
                               int damage, int heading, int seq, int effect,
                               int x, int y, int endX, int endY) {
        if (client == null || !client.isConnected()) return false;
        final CUserSocket connection = client.getConnection();
        if (connection == null || !connection.isClientConnected()) return false;
        final CIOFrame frame = connection.beginDirectFrame(39);
        if (frame == null) return false;
        boolean committed = false;
        try {
            final WriteBuffer buffer = frame.buffer();
            buffer.writeC(Opcodes.S_ATTACK);
            buffer.writeC(1);
            buffer.writeD(attackerId);
            buffer.writeD(targetId);
            buffer.writeH(damage);
            buffer.writeC(heading);
            buffer.writeD(seq);
            buffer.writeH(effect);
            buffer.writeC(0);
            buffer.writeH(x);
            buffer.writeH(y);
            buffer.writeH(endX);
            buffer.writeH(endY);
            buffer.writeD(0);
            buffer.writeH(0);
            buffer.writeC(0);
            buffer.writeH(0);
            committed = connection.commitDirectFrame(frame);
            return committed;
        } finally {
            if (!committed) connection.abortDirectFrame(frame);
        }
    }
}

                .order(java.nio.ByteOrder.LITTLE_ENDIAN);

        b.put((byte)Opcodes.S_ATTACK).put((byte)type);

            case MODE_2:

                 .put((byte)attackType);

                 .put((byte)heading).putShort((short)0).putShort((short)0)
                 .put((byte)attackType).putShort((short)gfx)
                 .putShort((short)0).putShort((short)0);
                break;

                throw new IllegalArgumentException("Unsupported attack mode "+mode);
        }
        return b.array();

private static boolean send0(
			CIOClient client,
			int mode,
			int chaId,
			int objId,
			int type,
			int heading,
			int attackType,
			int gfx) {

		}

		if (connection == null || !connection.isClientConnected()) {
			return false;
		}

		final int wireSize;

		switch (mode) {
		case MODE_1:
			wireSize = WIRE_SIZE_MODE_1;
			break;

		case MODE_2:
			wireSize = WIRE_SIZE_MODE_2;
			break;

		case MODE_3:
			wireSize = WIRE_SIZE_MODE_3;
			break;

		default:
			return false;
		}

		final CIOFrame frame =
				connection.beginDirectFrame(
						wireSize);

		if (frame == null) {
			return false;
		}

		final WriteBuffer buffer = frame.buffer();
		boolean committed = false;

		try {
			buffer.writeC(Opcodes.S_ATTACK);
			buffer.writeC(type);

			switch (mode) {
			case MODE_1:
				buffer.writeD(objId);
				buffer.writeD(chaId);
				buffer.writeC(0x4E);
				buffer.writeC(0);
				buffer.writeC(heading);
				buffer.writeD(0x00000000);
				buffer.writeC(0x00);
				break;

				buffer.writeH(0x0000);

				break;

				buffer.writeC(heading);

				buffer.writeH(0x0000);
				buffer.writeC(attackType);

import l1j.server.server.model.Instance.L1PcInstance;

import java.util.ArrayList;
import java.util.List;
import java.util.logging.Level;
import java.util.logging.Logger;
import l1j.server.MJKDASystem.MJKDA;
import l1j.server.MJKDASystem.MJKDALoadManager;

public final class SC_ChartRank {
    private static final Logger LOG = Logger.getLogger(SC_ChartRank.class.getName());
    private static final int SC_BASECAMP_CHART_NOTI_PACKET = 0x007F;
    private static final Snapshot ZERO = new Snapshot(new byte[0][], 0, true);

    private SC_ChartRank() {}

    public static final class Snapshot {
        private final byte[][] serialized;
        private final int serverPoint;
        private final boolean zero;
        private final int wireSize;

        private Snapshot(byte[][] serialized, int serverPoint, boolean zero) {
            this.serialized = serialized;
            this.serverPoint = serverPoint;
            this.zero = zero;
            long size = 2 + 3 + 1 + bitSize(serverPoint) + 2;
            if (zero) size = 2 + 3 + 1 + 1 + 2;
            else for (byte[] bytes : serialized) size += 1L + bitSize(bytes.length) + bytes.length;
            if (size > Integer.MAX_VALUE) throw new IllegalArgumentException("chart frame too large");
            this.wireSize = (int) size;
        }
    }

    private static int bitSize(long value) {
        if (value < 0L) value &= 0xFFFFFFFFL;
        int n = 1;
        while ((value & ~0x7FL) != 0) { n++; value >>>= 7; }
        return n;
    }

    public static Snapshot getZero() { return ZERO; }

    public static Snapshot get(List<MJKDA> list) {
        if (list == null || list.isEmpty()) return ZERO;
        final ArrayList<byte[]> bytes = new ArrayList<>(10);
        try {
            synchronized (list) {
                for (MJKDA kda : list) {
                    if (kda == null) continue;
                    if (kda.kill <= 0) break;
                    byte[] payload = kda.serialize();
                    if (payload != null && payload.length > 0) {
                        bytes.add(payload);
                        if (bytes.size() >= 10) break;
                    }
                }
            }
        } catch (Exception e) {
            LOG.log(Level.SEVERE, "SC_ChartRank KDA serialize", e);
        }
        if (bytes.isEmpty()) return ZERO;
        return new Snapshot(bytes.toArray(new byte[bytes.size()][]), MJKDALoadManager.KDA_TOTAL_PVP, false);
    }


    /** Freeze the exact existing direct SC_BASECAMP_CHART_NOTI_PACKET
     * payload once for a chart update rather than rewriting the same
     * Snapshot fields for every eligible viewer.
     */
    public static byte[] sharedWire(Snapshot snapshot) {
        if(snapshot==null || snapshot.wireSize>0xffff)return null;
        final java.nio.ByteBuffer out=java.nio.ByteBuffer.allocate(snapshot.wireSize)
                .order(java.nio.ByteOrder.LITTLE_ENDIAN);
        out.position(2);
        out.put((byte)Opcodes.S_EXTENDED_PROTOBUF);
        out.putShort((short)SC_BASECAMP_CHART_NOTI_PACKET);
        if(snapshot.zero){
            out.put((byte)0x10);
            out.put((byte)0);
            out.putShort((short)0);
        } else {
            for(byte[] payload:snapshot.serialized){
                out.put((byte)0x0a);
                putBit(out,payload.length);
                out.put(payload);
            }
            out.put((byte)0x10);
            putBit(out,snapshot.serverPoint);
            out.putShort((short)0);
        }
        if(out.position()!=snapshot.wireSize)
            throw new IllegalStateException("chart packet size mismatch");
        out.putShort(0,(short)snapshot.wireSize);
        return out.array();
    }

    /** WriteBuffer.writeBit uses 32-bit unsigned semantics for negatives. */
    private static void putBit(java.nio.ByteBuffer out,long value){
        if(value<0L)value &= 0xffffffffL;
        while((value & ~0x7fL)!=0L){
            out.put((byte)((value & 0x7fL)|0x80L));
            value >>>= 7;
        }
        out.put((byte)value);
    }

    public static boolean send(L1PcInstance pc, Snapshot snapshot) {
        if (pc == null || snapshot == null) return false;
        final CIOClient client = pc.getNetConnection();
        if (client == null || !client.isConnected()) return false;
        final CUserSocket connection = client.getConnection();
        if (connection == null || !connection.isClientConnected()) return false;
        final CIOFrame frame = connection.beginDirectFrame(snapshot.wireSize);
        if (frame == null) return false;
        boolean committed = false;
        try {
            final WriteBuffer out = frame.buffer();
            out.writeC(Opcodes.S_EXTENDED_PROTOBUF);
            out.writeH(SC_BASECAMP_CHART_NOTI_PACKET);
            if (snapshot.zero) {
                out.writeC(0x10);
                out.writeC(0);
                out.writeH(0);
            } else {
                for (byte[] bytes : snapshot.serialized) {
                    out.writeC(0x0A);
                    out.writeBit(bytes.length);
                    out.writeByte(bytes);
                }
                out.writeC(0x10);
                out.writeBit(snapshot.serverPoint);
                out.writeH(0);
            }
            committed = connection.commitDirectFrame(frame);
            return committed;
        } finally {
            if (!committed) connection.abortDirectFrame(frame);
        }
    }
}

     * applies receiver-specific encryption after enqueue.
     */
    public static byte[] sharedWire(int opcode,int type,int objectId,
                                    String text,boolean includeObjectId) {
        final byte[] txt=(text==null?"":text).getBytes(
                java.nio.charset.Charset.forName("MS949"));
        final long needed=5L+txt.length+(includeObjectId?4L:0L);
        if(needed>0xffffL)return null;
        final int size=(int)needed;
        final byte[] wire=new byte[size];
        wire[0]=(byte)size;
        wire[1]=(byte)(size>>>8);
        wire[2]=(byte)opcode;
        wire[3]=(byte)type;
        int at=4;
        if(includeObjectId){
            wire[at++]=(byte)objectId;
            wire[at++]=(byte)(objectId>>>8);
            wire[at++]=(byte)(objectId>>>16);
            wire[at++]=(byte)(objectId>>>24);

        System.arraycopy(txt,0,wire,at,txt.length);

    private static byte[] sharedDirectAdmissionWire(int opcode, int type,

    }

    private static boolean sendDirectRaw(L1PcInstance recipient, int mode, int opcode,
            int type, int objectId, String text1, String text2, int x, int y) {
        if (recipient == null || mode == 0) return false;
        final CIOClient socket = recipient.getNetConnection();
        if (socket == null || !socket.isConnected()) return false;
        final CUserSocket c = socket.getConnection();
        if (c == null || !c.isClientConnected()) return false;
        final long capacity = 32L + 4L * ((long) text1.length() + text2.length());
        if (capacity > Integer.MAX_VALUE) return false;
        final CIOFrame frame = c.beginDirectFrame((int) capacity);
        if (frame == null) return false;
        boolean committed = false;
        try {
            final WriteBuffer buffer = frame.buffer();
            writePayload(buffer, mode, opcode, type, objectId, text1, text2, x, y);
            committed = c.commitDirectFrame(frame);
            return committed;
        } finally {
            if (!committed) c.abortDirectFrame(frame);
        }
    }

    private static boolean sendInternal(L1PcInstance pc, String message, int index, MJSimpleRgb color,
                                        int surface, int duration, boolean indexMode) {
        if (pc == null) return false;
        final CIOClient socket = pc.getNetConnection();
        if (socket == null || !socket.isConnected()) return false;
        final CUserSocket connection = socket.getConnection();
        if (connection == null || !connection.isClientConnected()) return false;
        final String msg = message == null ? "" : message;
        final byte[] rgb = color == null ? null : color.get_bytes();
        final long needed = 64L + (indexMode ? 0L : ((long) msg.length() << 2))
                + (rgb == null ? 0 : rgb.length);
        if (needed > Integer.MAX_VALUE) return false;
        final CIOFrame frame = connection.beginDirectFrame((int) needed);
        if (frame == null) return false;
        boolean committed = false;
        try {
            writePayload(frame.buffer(), msg, index, rgb, surface, duration, indexMode);
            committed = connection.commitDirectFrame(frame);
            return committed;

        }

            buffer.writeBit(duration);

            buffer.writeBit(surface * 2);
            buffer.writeBit(0x12);
            buffer.writeS2(msg);
            buffer.writeBit(0x1A);
            if (rgb != null) buffer.writeByte(rgb);
            buffer.writeBit(0x20);
            buffer.writeBit(duration);
        }

            b.put((byte)((value&0x7fL)|0x80L));
            value>>>=7;
        }
        b.put((byte)value);
    }

    private static byte[] sharedWire(String msg,int index,MJSimpleRgb color,
            int surface,int duration,boolean indexMode) {
        final String text=msg==null?"":msg;
        final byte[] rgb=color==null?null:color.get_bytes();

        // Source's writeS2() rejects encoded lengths above 255. Fall back

            putVar(b,0x12);

            putVar(b,0x1a);
            if(rgb!=null)b.put(rgb);
            putVar(b,0x20);putVar(b,duration);
        }

    private static int maxBytes(String s) {
        return s == null ? 1 : (s.length() * 2) + 1;
    }
    private static boolean send(L1PcInstance pc, int idx, int type, String msg1, String msg2, boolean two) {
        if (pc == null) return false;
        final CIOClient socket = pc.getNetConnection();
        if (socket == null || !socket.isConnected()) return false;
        final CUserSocket conn = socket.getConnection();
        if (conn == null || !conn.isClientConnected()) return false;
        final long capacity = 16L + maxBytes(msg1) + (two ? maxBytes(msg2) : 0);
        if (capacity > Integer.MAX_VALUE) return false;
        final CIOFrame frame = conn.beginDirectFrame((int) capacity);
        if (frame == null) return false;
        boolean committed = false;
        try {
            final WriteBuffer buffer = frame.buffer();
            buffer.writeC(Opcodes.S_ASK);
            buffer.writeH(0);
            buffer.writeD(idx);
            buffer.writeH(type);
            buffer.writeS(msg1 == null ? "" : msg1);
            if (two) buffer.writeS(msg2 == null ? "" : msg2);
            committed = conn.commitDirectFrame(frame);
            return committed;
        } finally {
            if (!committed) conn.abortDirectFrame(frame);
        }
    }

        if(wire==null)return;

    }

    public static byte[] sharedIndexedWire(int idx,int type,String message) {
        final byte[] text=(message==null?"":message)

        return b.array();
    }

    private static boolean sendFrame(L1PcInstance receiver, int type, int chatType,
                                     byte[] textBytes, byte[] targetNameBytes,
                                     byte[] senderNameBytes, int pcObjId, int pcX,
                                     int pcY, int rankLevel) {

        if (connection == null || !connection.isClientConnected()) return false;

        if (frame == null) return false;

	private static void writePayload(WriteBuffer buffer, int type, int chatType, byte[] textBytes, byte[] targetNameBytes, byte[] senderNameBytes, int pcObjId, int pcX, int pcY, int rankLevel) {
		buffer.writeC(Opcodes.S_EXTENDED_PROTOBUF);

		switch (type) {
			case 3:
				buffer.writeC(0x03);
				break;
			case 4:
				buffer.writeC(0x04);
				break;
		}

		buffer.writeC(0x02);
		buffer.writeC(0x08);
		buffer.writeC(0x00);
		buffer.writeC(0x10);
		buffer.writeC(chatType);

		buffer.writeC(0x1a);
		buffer.writeC(textBytes.length);
		buffer.writeByte(textBytes);

		switch (type) {
			case 3:
				buffer.writeC(0x22);
				if (chatType == 0) {
					buffer.writeC(0x00);
					buffer.writeC(0x28);
					buffer.writeC(0x00);
					buffer.writeC(0x30);
					buffer.writeC(0x18);
					buffer.writeC(0x38);
					buffer.writeC(0x00);
				} else if (chatType == 1) {
					buffer.writeC(targetNameBytes.length);
					buffer.writeByte(targetNameBytes);
					buffer.writeC(0x28);
					buffer.writeC(0x00);
					buffer.writeC(0x30);
					buffer.writeC(0x00);
					buffer.writeH(0);
				}
				break;
			case 4:
				buffer.writeC(0x2a);
				buffer.writeC(senderNameBytes.length);
				buffer.writeByte(senderNameBytes);

				if (chatType == 0) {
					buffer.writeC(0x38);
					buffer.writeK(pcObjId);
					buffer.writeC(0x40);
					buffer.writeK(pcX);
					buffer.writeC(0x48);
					buffer.writeK(pcY);
				}
				if (rankLevel != 0) {
					buffer.writeC(0x50);
					buffer.writeC(rankLevel);
				}
				buffer.writeH(0);
				break;
		}
	}

    /** Preserves the historical writeK(int) (not generic protobuf varint).
     * In particular it uses signed int division and Java's 5-byte path.
     */
    private static void putK(java.nio.ByteBuffer b,int value) {
        final int v=value/128;
        if(v==0)b.put((byte)value);
        else if(v<=127)b.put((byte)((value&0x7f)+128)).put((byte)v);
        else if(v<=16383)b.put((byte)((value&0x7f)+128))
                .put((byte)((v&0x7f)+128)).put((byte)(v/128));
        else if(v<=2097151)b.put((byte)((value&0x7f)+128))
                .put((byte)((v&0x7f)+128)).put((byte)(((v/128)&0x7f)+128))
                .put((byte)(v/16384));
        else b.put((byte)((value&0x7f)+128))
                .put((byte)((v&0x7f)+128)).put((byte)(((v/128)&0x7f)+128))
                .put((byte)(((v/16384)&0x7f)+128)).put((byte)(v/2097152));
    }

    public static byte[] sharedWire(ChatData d) {
        final long capacity=2L+64L+d.textBytes.length+d.targetBytes.length
                              +d.senderBytes.length;
        if(capacity>0xFFFF)return null;
        final java.nio.ByteBuffer b=java.nio.ByteBuffer.allocate((int)capacity)
                .order(java.nio.ByteOrder.LITTLE_ENDIAN);
        b.position(2);
        b.put((byte)Opcodes.S_EXTENDED_PROTOBUF);
        if(d.type==3)b.put((byte)0x03);
        else if(d.type==4)b.put((byte)0x04);
        b.put((byte)0x02).put((byte)0x08).put((byte)0x00)
         .put((byte)0x10).put((byte)d.chatType);
        b.put((byte)0x1a).put((byte)d.textBytes.length).put(d.textBytes);
        if(d.type==3) {
            b.put((byte)0x22);
            if(d.chatType==0)
                b.put((byte)0x00).put((byte)0x28).put((byte)0x00)
                 .put((byte)0x30).put((byte)0x18).put((byte)0x38).put((byte)0x00);
            else if(d.chatType==1)
                b.put((byte)d.targetBytes.length).put(d.targetBytes)
                 .put((byte)0x28).put((byte)0x00).put((byte)0x30)
                 .put((byte)0x00).putShort((short)0);
        } else if(d.type==4) {
            b.put((byte)0x2a).put((byte)d.senderBytes.length).put(d.senderBytes);
            if(d.chatType==0) {
                b.put((byte)0x38);putK(b,d.pcObjId);
                b.put((byte)0x40);putK(b,d.pcX);
                b.put((byte)0x48);putK(b,d.pcY);
            }
            if(d.rankLevel!=0)b.put((byte)0x50).put((byte)d.rankLevel);
            b.putShort((short)0);
        }
        final int size=b.position();
        if(size>0xFFFF)return null;
        b.putShort(0,(short)size);
        return java.util.Arrays.copyOf(b.array(),size);
    }

    private static String formatNpcChat(L1NpcInstance npc, String chat, int type) {
        if (npc == null || chat == null) return null;
        switch (type) {
        case 0:
        case 4:
            return npc.getNpcId() == 8500200 ? chat : npc.getName() + ": " + chat;
        case 2:
            int id = npc.getNpcTemplate() != null ? npc.getNpcTemplate().get_npcId() : 0;
            return id == 70518 || id == 70506
                    ? npc.getName() + ": " + chat : "<" + npc.getName() + "> " + chat;
        case 3:
            return "[" + npc.getName() + "] " + chat;
        default:
            return null;
        }
    }


    /** Identical S_SAY_CODE payload for all viewers; socket-specific cipher
     * is applied by the existing native FIFO, never to this shared plaintext.
     */
    public static byte[] sharedWire(int objectId,int color,String text) {
        if(text==null)return null;
        final byte[] chars=text.getBytes(java.nio.charset.Charset.forName("MS949"));
        final int size=9+chars.length;
        if(size>0xffff)return null;
        final byte[] wire=new byte[size];
        wire[0]=(byte)size;
        wire[1]=(byte)(size>>>8);
        wire[2]=(byte)Opcodes.S_SAY_CODE;
        wire[3]=(byte)color;
        wire[4]=(byte)objectId;
        wire[5]=(byte)(objectId>>>8);
        wire[6]=(byte)(objectId>>>16);
        wire[7]=(byte)(objectId>>>24);
        System.arraycopy(chars,0,wire,8,chars.length);
        return wire;
    }

    private static boolean sendFormatted(L1PcInstance pc, int npcId, int color, String text) {
        if (pc == null || text == null) return false;
        final CIOClient socket = pc.getNetConnection();
        if (socket == null || !socket.isConnected()) return false;
        final CUserSocket connection = socket.getConnection();
        if (connection == null || !connection.isClientConnected()) return false;
        final long capacity = 32L + 4L * text.length();
        if (capacity > Integer.MAX_VALUE) return false;
        final CIOFrame frame = connection.beginDirectFrame((int) capacity);
        if (frame == null) return false;
        boolean committed = false;
        try {
            final WriteBuffer buffer = frame.buffer();
            buffer.writeC(Opcodes.S_SAY_CODE);
            buffer.writeC(color);
            buffer.writeD(npcId);
            buffer.writeS(text);
            committed = connection.commitDirectFrame(frame);
            return committed;
        } finally {
            if (!committed) connection.abortDirectFrame(frame);
        }
    }

    public static boolean sendCharacter(L1PcInstance pc, L1Character cha, String chat) {
        if (cha == null) return false;
        return sendFormatted(pc, cha.getId(), 0, chat == null ? "" : chat);
    }

        final String message=chat==null?"":chat;

                L1World.getInstance().getVisiblePlayer(source),

    public static void broadcastNpcFast(L1Character source, L1NpcInstance npc, String chat, int type) {
        if(source==null || npc==null)return;
        final String formatted=formatNpcChat(npc,chat,type);
        if(formatted==null)return;
        final int color=type==4?NPC_CHAT_NOMAL_4:type;
        final byte[] wire=sharedWire(npc.getId(),color,formatted);
        final L1World world=L1World.getInstance();
        final java.util.List<L1PcInstance> visible=world.borrowVisiblePlayerFast(source);
        try {
            l1j.server.server.model.network.core.broadcast.CSharedBroadcast.sendRecipients(visible,wire);
        } finally {

    private static void dispatch(L1Character caster, L1Character[] targets, int count,
                                 int gfx, int action, int type, boolean toSelf) {
        if (caster == null) return;
        final int x = caster.getX(), y = caster.getY(), id = caster.getId();
        final int limit = targets == null || count <= 0 ? 0 : Math.min(count, targets.length);
        int heading = caster.getHeading();
        if (type == TYPE_DIR) {
            for (int i = 0; i < limit; i++) {
                final L1Character t = targets[i];
                if (t == null) continue;
                heading = calcHeading(x, y, t.getX(), t.getY());
                try { caster.setHeading(heading); } catch (Throwable ignored) {}
                break;
            }
        }
        final int[] ids = limit <= 0 ? EMPTY : new int[limit];
        int used = 0;
        for (int i = 0; i < limit; i++) {
            final L1Character t = targets[i];
            if (t != null) ids[used++] = t.getId();
        }
        final int seq = SEQUENCE.incrementAndGet();
        if (toSelf && caster instanceof L1PcInstance) {
            send((L1PcInstance) caster, id, x, y, heading, seq, gfx, action, type, ids, used);
        }
        final int frozenHeading=heading, frozenCount=used;
        final byte[] wire=sharedWire(id,x,y,frozenHeading,seq,gfx,action,type,ids,frozenCount);
        if(wire!=null) {
            l1j.server.server.model.network.core.broadcast.CSharedBroadcast.send(caster, wire, true, -1, null);
        }
}

    private static int calcHeading(int x, int y, int tx, int ty) {
        if (tx > x) return ty > y ? 3 : ty < y ? 1 : 2;
        if (tx < x) return ty > y ? 5 : ty < y ? 7 : 6;
        if (ty < y) return 0;
        if (ty > y) return 4;
        return 0;
    }

    private static boolean send(L1PcInstance pc, int id, int x, int y, int heading,
                                int seq, int gfx, int action, int type, int[] ids, int count) {
        if (pc == null) return false;
        final CIOClient client = pc.getNetConnection();
        if (client == null || !client.isConnected()) return false;
        final CUserSocket connection = client.getConnection();
        if (connection == null || !connection.isClientConnected()) return false;
        final long size = 24L + (long) count * 6L;
        if (size > Integer.MAX_VALUE) return false;
        final CIOFrame frame = connection.beginDirectFrame((int) size);
        if (frame == null) return false;
        boolean committed = false;
        try {
            final WriteBuffer b = frame.buffer();
            b.writeC(Opcodes.S_ATTACK_MANY);
            b.writeC(action);
            b.writeD(id);
            b.writeH(x);
            b.writeH(y);
            b.writeC(heading);
            b.writeD(seq);
            b.writeH(gfx);
            b.writeC(type);
            b.writeH(0);
            b.writeH(count);
            for (int i = 0; i < count; i++) {
                b.writeD(ids[i]);
                b.writeC(32);
                b.writeC(action);
            }
            committed = connection.commitDirectFrame(frame);
            return committed;
        } finally {
            if (!committed) connection.abortDirectFrame(frame);
        }
    }

         .putInt(id).putShort((short)x).putShort((short)y).put((byte)heading)
         .putInt(seq).putShort((short)gfx).put((byte)type)

        if(message==null)return sharedWire(type);
        final byte[] encoded=message.getBytes(
                java.nio.charset.Charset.forName("MS949"));
        final int length=7+encoded.length;
        if(length>0xffff)return null;
        final byte[] wire=new byte[length];
        wire[0]=(byte)length;
        wire[1]=(byte)(length>>>8);
        wire[2]=(byte)Opcodes.S_MESSAGE_CODE;
        wire[3]=(byte)type;
        wire[4]=(byte)(type>>>8);
        wire[5]=1;
        System.arraycopy(encoded,0,wire,6,encoded.length);
        return wire;
    }

        final byte[] b2=msg2==null?null:msg2.getBytes(charset);
        final byte[] b3=msg3==null?null:msg3.getBytes(charset);

	private static boolean send0(
			CIOClient client,
			int actionId,
			int chaId,
			int targetObjId,
			int isHit,
			int heading,
			int seq,
			int spellGfx,
			int myX,
			int myY,
			int targetX,
			int targetY) {

		if (client == null || !client.isConnected()) {
			return false;
		}

		final CUserSocket connection =
				client.getConnection();

		if (connection == null
				|| !connection.isClientConnected()) {
			return false;
		}


		final CIOFrame frame =
				connection.beginDirectFrame(
						WIRE_SIZE);

		if (frame == null) {
			return false;
		}

		final WriteBuffer buffer =
				frame.buffer();

		boolean committed = false;

		try {
			buffer.writeC(Opcodes.S_ATTACK);
			buffer.writeC(actionId);
			buffer.writeD(chaId);
			buffer.writeD(targetObjId);
			buffer.writeC(isHit);
			buffer.writeC(0);
			buffer.writeC(heading);
			buffer.writeD(seq);
			buffer.writeH(spellGfx);
			buffer.writeC(6);
			buffer.writeH(myX);
			buffer.writeH(myY);
			buffer.writeH(targetX);
			buffer.writeH(targetY);
			buffer.writeC(0);
			buffer.writeC(0);
			buffer.writeC(0);
			buffer.writeC(0);
			buffer.writeC(0);

			committed =
					connection.commitDirectFrame(
							frame);

			return committed;

		} finally {
			if (!committed) {
				connection.abortDirectFrame(frame);
			}
		}
	}

	private static int fixActionId(
			L1Character cha,
			int actionId) {

		int fixed = actionId;

		if (cha instanceof L1PcInstance) {
			if (cha.hasSkillEffect(
					L1SkillId.SHAPE_CHANGE)
					&& fixed
							== ActionCodes.ACTION_SkillAttack) {

				final int gfx =
						cha.getCurrentSpriteId();

				if (gfx == 5727
						|| gfx == 5730) {

					fixed =
							ActionCodes.ACTION_SkillBuff;

				} else if (gfx == 5733
						|| gfx == 5736) {

					fixed =
							ActionCodes.ACTION_Attack;
				}
			}
		}

		if (cha.equalsCurrentSprite(4013)) {
			fixed =
					ActionCodes.ACTION_Attack;
		}

		return fixed;
	}

	private static int calcHeading(
			int myX,
			int myY,
			int targetX,
			int targetY) {

		final int dx =
				Integer.compare(
						targetX,
						myX);

		final int dy =
				Integer.compare(
						targetY,
						myY);

		if (dx == 0 && dy < 0) {
			return 0;
		}

		if (dx > 0 && dy < 0) {
			return 1;
		}

		if (dx > 0 && dy == 0) {
			return 2;
		}

		if (dx > 0 && dy > 0) {
			return 3;
		}

		if (dx == 0 && dy > 0) {
			return 4;
		}

		if (dx < 0 && dy > 0) {
			return 5;
		}

		if (dx < 0 && dy == 0) {
			return 6;
		}

		if (dx < 0 && dy < 0) {
			return 7;
		}

		return 0;
	}


    private static byte[] sharedWire(int actionId,int chaId,int targetObjId,
            int isHit,int heading,int seq,int spellGfx,int myX,int myY,
            int targetX,int targetY) {
        final java.nio.ByteBuffer b=java.nio.ByteBuffer.allocate(WIRE_SIZE)
                .order(java.nio.ByteOrder.LITTLE_ENDIAN);
        b.putShort((short)WIRE_SIZE).put((byte)Opcodes.S_ATTACK).put((byte)actionId)
         .putInt(chaId).putInt(targetObjId).put((byte)isHit).put((byte)0)
         .put((byte)heading).putInt(seq).putShort((short)spellGfx).put((byte)6)
         .putShort((short)myX).putShort((short)myY)
         .putShort((short)targetX).putShort((short)targetY)
         .put((byte)0).put((byte)0).put((byte)0).put((byte)0).put((byte)0);
        return b.array();
    }

    public static boolean send(CIOClient client, int type, String clan1, String clan2) {
        if (client == null || !client.isConnected()) return false;
        final CUserSocket conn = client.getConnection();
        if (conn == null || !conn.isClientConnected()) return false;
        final CIOFrame frame = conn.beginDirectFrame(6 + ((clan1 == null ? 0 : clan1.length()) + (clan2 == null ? 0 : clan2.length())) * 2);
        if (frame == null) return false;
        boolean committed = false;
        try {
            final WriteBuffer buffer = frame.buffer();
            buffer.writeC(Opcodes.S_WAR);
            buffer.writeC(type);
            buffer.writeS(clan1 == null ? "" : clan1);
            buffer.writeS(clan2 == null ? "" : clan2);

        } finally {

        }

    }
    public static void broadcastClan(l1j.server.server.model.L1Clan clan, int type, String clan1, String clan2) {
        if (clan == null) return;
        final byte[] wire=sharedWire(type, clan1, clan2);
        if(wire==null)return;
        try (var shared = new l1j.server.server.model.network.core.broadcast.CAsyncSendConstBuffer(

            });
        }
    }

        final java.nio.ByteBuffer out=java.nio.ByteBuffer.allocate((int)size)
                .order(java.nio.ByteOrder.LITTLE_ENDIAN);

           .put(a).put((byte)0).put(b).put((byte)0);

public final class SC_WarStartMessage {
    private static final String[] names = {
        "켄트성", "오크 요새", "윈다우드성", "기란성", "하이네성", "지저성", "아덴성"
    };
    private SC_WarStartMessage() {}

    public static String[] snapshot() {
        synchronized (names) {
            for (L1Clan clan : L1World.getInstance().getAllClans()) {
                final int index = clan.getCastleId() - 1;
                if (index > 0 && index < names.length) names[index] = clan.getClanName();
            }
            return names.clone();
        }
    }


    /** Immutable full S_EVENT 0x4e castle roster. Seven nul-terminated MS949
     * castle names in the snapshot's original order, including null as "".
     */
    public static byte[] sharedWire(String[] clanNames) {
        if(clanNames==null)throw new IllegalArgumentException("clan names");
        final java.nio.charset.Charset ms949=java.nio.charset.Charset.forName("MS949");
        int size=5;
        for(String name:clanNames){
            final int len=name==null?0:name.getBytes(ms949).length;
            size+=len+1;
            if(size>65535)throw new IllegalStateException("castle roster overflow");
        }
        final byte[] wire=new byte[size];
        wire[0]=(byte)size;wire[1]=(byte)(size>>>8);
        wire[2]=(byte)Opcodes.S_EVENT;wire[3]=0x4e;wire[4]=0x07;
        int at=5;
        for(String name:clanNames){
            if(name!=null){
                final byte[] v=name.getBytes(ms949);
                System.arraycopy(v,0,wire,at,v.length);
                at+=v.length;
            }
            at++;
        }
        if(at!=wire.length)throw new IllegalStateException("roster wire length");
        return wire;
    }

    public static boolean send(L1PcInstance pc, String[] clanNames) {
        if (pc == null || pc.getNetConnection() == null || !pc.getNetConnection().isConnected()) return false;
        final CUserSocket conn = pc.getNetConnection().getConnection();
        if (conn == null || !conn.isClientConnected()) return false;
        int length = 5;
        for (String name : clanNames) length += 1 + (name == null ? 0 : name.length() * 2);
        final CIOFrame frame = conn.beginDirectFrame(length);
        if (frame == null) return false;
        boolean committed = false;
        try {
            final WriteBuffer buffer = frame.buffer();
            buffer.writeC(Opcodes.S_EVENT);
            buffer.writeC(0x4E);
            buffer.writeC(0x07);
            for (String name : clanNames) buffer.writeS(name);

public final class SC_Weather {
    private SC_Weather() {}

    public static boolean send(CIOClient client, int weather) {
        if (client == null || !client.isConnected()) return false;
        final CUserSocket connection = client.getConnection();
        if (connection == null || !connection.isClientConnected()) return false;
        final CIOFrame frame = connection.beginDirectFrame(4);
        if (frame == null) return false;
        boolean committed = false;
        try {
            final WriteBuffer buffer = frame.buffer();
            buffer.writeC(Opcodes.S_WEATHER);
            buffer.writeC(weather);
            committed = connection.commitDirectFrame(frame);
            return committed;
        } finally {
            if (!committed) connection.abortDirectFrame(frame);
        }
    }

    public static boolean send(L1PcInstance pc, int weather) {
        return pc != null && send(pc.getNetConnection(), weather);
    }

    public static void broadcastAll(int weather) {
        final byte[] wire=sharedWire(weather);
        
        l1j.server.server.model.network.core.broadcast.CSharedBroadcast.sendAll(wire);
    }


    private static byte[] sharedWire(int weather) {
        final java.nio.ByteBuffer b=java.nio.ByteBuffer.allocate(4)
                .order(java.nio.ByteOrder.LITTLE_ENDIAN);
        b.putShort((short)4).put((byte)Opcodes.S_WEATHER).put((byte)weather);
        return b.array();
    }

    public static boolean sendPcFull(
            CIOClient client,
            L1PcInstance pc,
            String requestedName,
            boolean revealInvisibleName) {

        if (pc == null
                || client == null
                || !client.isConnected()) {
            return false;
        }

        final CUserSocket connection =
                client.getConnection();

        if (connection == null
                || !connection.isClientConnected()) {
            return false;
        }

        /* Capture once. No snapshot object. */
        final int objectId = pc.getId();
        if (objectId == 0) {
            return false;
        }

        final int x = pc.getX();
        final int y = pc.getY();
        final int heading = pc.getHeading();
        final int spriteId = pc.getCurrentSpriteId();
        final int action = computePcAction(pc);

        final int lightRadius = computePcLightRadius(pc);
        final int lawful = pc.getLawful();
        final int level = pc.getLevel();
        final int speedData = pc.isHaste() ? 1 : 0;
        final int emotion = computePcEmotion(pc);

        final boolean isGhost = pc.isGhost();
        final boolean isParalyzed = pc.isParalyzed();
        final boolean isInvisible = pc.isInvisble();
        final boolean isPoisoned = pc.getPoison() != null;

        final String desc =
                computePcDesc(
                        pc,
                        requestedName,
                        revealInvisibleName);

        final String title = safeString(getTitle(pc));
        final String clanName = safeString(pc.getClanname());
        final String masterName = "";

        final byte[] shopChat = pc.getShopChat();
        final int shopChatLength =
                shopChat != null ? shopChat.length : 0;

        final L1Clan clan = safeClan(pc);
        final boolean combatField = safeCombatField(pc);

        final int emblemId =
                clan != null && !combatField
                        ? safeClanEmblem(clan)
                        : 0;

        final int boundaryIndex =
                safeBoundaryIndex(level, spriteId);

        final int markStatus =
                computePcMarkStatus(pc, combatField);

        final boolean isCastleOffense =
                clan != null
                        && safeClanRank(pc) == L1Clan.군주
                        && safeCastleOffense(clan);

        final int reserve = complexReserve(
                shopChatLength,
                desc,
                title,
                clanName,
                masterName);

        if (reserve <= 0) {
            return false;
        }


        final CIOFrame frame =
                connection.beginDirectFrame(
                        reserve);

        if (frame == null) {
            return false;
        }

                    b,
                    x, y,
                    objectId,
                    spriteId,
                    action,
                    heading,
                    lightRadius,

                    lawful,
                    desc,
                    title,
                    speedData,
                    emotion);

            writeBool(b, 14, isGhost);
            writeBool(b, 15, isParalyzed);
            writeBool(b, 16, true);
            writeBool(b, 17, isInvisible);
            writeBool(b, 18, isPoisoned);

            writeVarint(b, 23, -1);

    public static boolean sendRaw(
            CIOClient client,
            byte[] body) {

        if (body == null
                || body.length == 0
                || client == null
                || !client.isConnected()) {
            return false;
        }

        final long reserveLong =
                (long) ConnectionConfig.HEADER_SIZE
                        + 1L
                        + Short.BYTES
                        + body.length;

        if (reserveLong > ConnectionConfig.MAX_WIRE_PACKET_SIZE) {
            return false;
        }

        final CUserSocket connection =
                client.getConnection();

        if (connection == null
                || !connection.isClientConnected()) {
            return false;
        }


        final CIOFrame frame =
                connection.beginDirectFrame(
                        (int) reserveLong);

        if (frame == null) {
            return false;
        }

        final WriteBuffer b = frame.buffer();

        boolean committed = false;

        try {
            writeHeader(b);
            b.writeByte(body);

            committed =
                    connection.commitDirectFrame(
                            frame);

            return committed;

        } catch (Throwable ignored) {
            return false;

        } finally {
            if (!committed) {
                connection.abortDirectFrame(frame);
            }
        }
    }

        l1j.server.server.model.network.core.broadcast.CSharedBroadcast.send(anchor, frozen, true, -1, null);
    }

    private static int complexReserve(
            int rawBytes,
            String a,
            String b,
            String c,
            String d) {

        long reserve =
                (long) FIXED_COMPLEX_RESERVE
                        + Math.max(rawBytes, 0);

        reserve += maxMs949Bytes(a);
        reserve += maxMs949Bytes(b);
        reserve += maxMs949Bytes(c);
        reserve += maxMs949Bytes(d);

        if (reserve > ConnectionConfig.MAX_WIRE_PACKET_SIZE) {
            return -1;
        }

        return (int) reserve;
    }

    private static long maxMs949Bytes(String value) {
        return value == null
                ? 0L
                : ((long) value.length() << 1);
    }

    private static void writeHeader(
            WriteBuffer b) {

        b.writeC(Opcodes.S_EXTENDED_PROTOBUF);
        b.writeH(SC_WORLD_PUT_OBJECT_NOTI);
    }

    private static void writeBase(
            WriteBuffer b,
            int x,
            int y,
            int objectId,
            int spriteId,
            int action,
            int heading,
            int lightRadius,
            int objectCount,
            int lawful,
            String desc,
            String title,
            int speedData,
            int emotion) {

        writePoint(b, 1, x, y);
        writeVarint(b, 2, objectId);
        writeVarint(b, 3, spriteId);
        writeVarint(b, 4, action);
        writeVarint(b, 5, heading);
        writeVarint(b, 6, lightRadius);
        writeVarint(b, 7, objectCount);
        writeVarint(b, 8, lawful);
        writeString(b, 9, desc);
        writeString(b, 10, title);
        writeVarint(b, 11, speedData);
        writeVarint(b, 12, emotion);
    }

    private static void writePoint(
            WriteBuffer b,
            int field,
            int x,
            int y) {

        b.writeBit((long) field << 3);
        b.writePoint(x, y);
    }

    private static void writeVarint(
            WriteBuffer b,
            int field,
            long value) {

        b.writeBit((long) field << 3);
        b.writeBit(value);
    }

    private static void writeBool(
            WriteBuffer b,
            int field,
            boolean value) {

        b.writeBit((long) field << 3);
        b.writeB(value);
    }

    private static void writeString(
            WriteBuffer b,
            int field,
            String value) {

        b.writeBit(((long) field << 3) | 2L);
        b.writeS2(safeString(value));
    }

    private static void writeByteArray(
            WriteBuffer b,
            int field,
            byte[] value) {

        b.writeBit(((long) field << 3) | 2L);

        if (value == null || value.length == 0) {
            b.writeBit(0);
            return;
        }

        b.writeBit(value.length);
        b.writeByte(value);
    }

    private static void writeTrailer(
            WriteBuffer b) {

        b.writeH(0);
    }

    private static int computePcAction(
            L1PcInstance pc) {

        if (pc.isDead()) {
            return ActionCodes.ACTION_Die;
        }

        if (pc.isPrivateShop()) {
            return 70;
        }

        if (pc.isFishing()) {
            return 71;
        }

        return pc.getCurrentWeapon();
    }

    private static int computePcEmotion(
            L1PcInstance pc) {

        if (pc.isBrave()
                || pc.isBlood_lust()
                || pc.isElfBraveMagicShort()) {
            return 0x01;
        }

        if (pc.isElfBrave()) {
            return 0x03;
        }

        if (pc.hasSkillEffect(L1SkillId.맥시멈)
                && pc.hasSkillEffect(L1SkillId.STATUS_ELFBRAVE)) {
            return 0x03;
        }

        if (pc.hasSkillEffect(L1SkillId.에볼루션)
                && pc.hasSkillEffect(L1SkillId.STATUS_ELFBRAVE)) {
            return 0x03;
        }

        if (pc.isPassive(MJPassiveID.DARK_HORSE.toInt())
                && pc.hasSkillEffect(L1SkillId.STATUS_FRUIT)) {
            return 0x03;
        }

        if (pc.isFastMovable() || pc.isFruit()) {
            return 0x04;
        }

        if (pc.isElfBraveMagicLong()) {
            return 0x09;
        }

        if (pc.istest()) {
            return pc.get_focus_wave_level() + 9;
        }

        return 0;
    }

    private static int computeNpcAction(
            L1NpcInstance npc) {

        final int npcId = npc.getNpcId();
        final int spriteId = npc.getCurrentSpriteId();

        if (npc.getShopName() != null) {
            return 70;
        }

        if ((npc.getNpcTemplate().is_doppel() && spriteId != 31)
                || spriteId == 727
                || spriteId == 985
                || spriteId == 986
                || spriteId == 6632
                || spriteId == 6634
                || spriteId == 6636
                || spriteId == 6638) {
            return 4;
        }

        if (spriteId == 110) {
            return 24;
        }

        if (spriteId == 111) {
            return 4;
        }

        if (spriteId == 51 || npcId == 60519) {
            return 24;
        }

        if (spriteId == 816) {
            return 20;
        }

        if (npcId == 203053) {
            return 4;
        }

        if (npcId == 203052) {
            return 20;
        }

        return npc.getStatus();
    }

    private static String computeNpcDesc(

        if (npc.getNpcId() == 6200008) {
            try {
                return safeString(
                        ((L1MerchantInstance) npc)
                                .getClanname());
            } catch (RuntimeException e) {
                return "";
            }
        }

        if (npc instanceof L1SignboardInstance) {
            return "";
        }

        final String nameId = npc.getNameId();

                ? nameId
                : "";
    }

        return safeString(npc.getTitle());

    private static byte[] sharedWireNpc(L1NpcInstance npc) {
        if(npc==null)return null;
        try {
            /* Capture once. No NpcSnapshot allocation. */
        final int objectId = npc.getId();
        if (objectId == 0) {
            return null;
        }

        final int x = npc.getX();
        final int y = npc.getY();
        final int spriteId = npc.getCurrentSpriteId();
        final int action = computeNpcAction(npc);
        final int heading = npc.getHeading();

        final int lightSize =
                npc.getNpcTemplate().getLightSize();

        final int lawful = npc.getTempLawful();
        final String desc = computeNpcDesc(npc);
        final String title = computeNpcTitle(npc);
        final int moveSpeed = npc.getMoveSpeed();
        final int braveSpeed = npc.getBraveSpeed();
        final int emblemId = computeNpcEmblem(npc);
        final String shopName = npc.getShopName();
        final int teamId = computeNpcTeamId(npc);

        final boolean isParalyzed =
                npc.getParalysis() != null;

        final boolean isPassObject = npc.isPassObject();
        final boolean isInvisible = npc.isInvisble();
        final boolean isPoisoned = npc.getPoison() != null;

        final L1Character master = npc.getMaster();
        final String masterName =
                master != null
                        ? safeString(master.getName())
                        : "";

        final int level = npc.getLevel();
        final int weakAttr =
                npc.getNpcTemplate().get_weakAttr();

        final long explosionTime = npc.getExplosionTime();
        final int npcClassId = npc instanceof L1MonsterInstance
                ? npc.getNpcTemplate().get_npc_class_id() : 0;

        final int reserve = complexReserve(
                0,
                desc,
                title,
                masterName,
                shopName);

        


        
            final int capacity=reserve>0 && reserve<=0xFFFF?reserve:0xFFFF;
            final java.nio.ByteBuffer bytes=java.nio.ByteBuffer.allocate(capacity)
                    .order(java.nio.ByteOrder.LITTLE_ENDIAN);
            final WriteBuffer b=new WriteBuffer();
            b.armExternal(bytes,0,capacity);
            b.beginDirect();
            b.position(ConnectionConfig.HEADER_SIZE);
            try {
                writeHeader(b);

            writeBase(
                    b,
                    x, y,
                    objectId,
                    spriteId,
                    action,
                    heading,
                    lightSize,
                    1,
                    lawful,
                    desc,
                    title,
                    moveSpeed,
                    braveSpeed);

            writeVarint(b, 13, 0);
            writeBool(b, 14, false);
            writeBool(b, 15, isParalyzed);
            writeBool(
                    b,
                    16,
                    shopName != null || !isPassObject);
            writeBool(b, 17, isInvisible);
            writeBool(b, 18, isPoisoned);

            writeVarint(b, 19, emblemId);
            writeString(b, 20, "");
            writeString(b, 21, masterName);
            writeVarint(b, 22, 5);
            writeVarint(b, 23, -1);
            writeVarint(b, 24, level);

            if (shopName != null) {
                writeString(b, 25, shopName);
            }

            writeVarint(b, 26, 1);
            writeBool(b, 27, false);
            writeVarint(b, 28, 0);
            writeVarint(b, 29, weakAttr);
            writeVarint(b, 30, -1);
            writeVarint(b, 31, 0);
            writeVarint(b, 32, 0);
            writeVarint(b, 33, teamId);
            writeVarint(b, 34, 0);
            writeVarint(b, 35, 0);
            writeVarint(b, 36, 0);

            if (explosionTime != 0L) {
                writeVarint(b, 37, explosionTime);
            }

            if (npcClassId > 0) {
                writeVarint(b, 39, npcClassId);
            }

            writeTrailer(b);
                final int length=b.position();
                if(length>0xFFFF)return null;
                bytes.putShort(0,(short)length);
                return java.util.Arrays.copyOf(bytes.array(),length);
            } finally {
                b.releaseResources();
            }
        } catch(Throwable ignored) {
            return null;
        }
    }
